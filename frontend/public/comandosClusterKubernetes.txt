#Actualizar las Raspberry 
sudo update
sudo upgrade -y

#Desactivar el Swap
sudo swapoff -a

#Cargar los módulos del kernel y configurar Sysctl
sudo modprobe overlay
sudo modprobe br_netfilter

# Añadir los módulos para que se carguen al inicio
echo "overlay" | sudo tee -a /etc/modules-load.d/k8s.conf
echo "br_netfilter" | sudo tee -a /etc/modules-load.d/k8s.conf

# Configurar sysctl para el paso de tráfico de red
sudo tee /etc/sysctl.d/k8s.conf <<EOF
net.bridge.bridge-nf-call-ip6tables = 1
net.bridge.bridge-nf-call-iptables = 1
net.ipv4.ip_forward = 1
EOF

# Aplicar la configuración de sysctl
sudo sysctl --system

# Instalar dependencias para Docker GPG key
sudo apt install -y apt-transport-https ca-certificates curl gnupg-agent software-properties-common

# Añadir la clave GPG oficial de Docker
curl -fsSL https://download.docker.com/linux/ubuntu/gpg | sudo gpg --dearmor -o /etc/apt/keyrings/docker.gpg
sudo chmod a+r /etc/apt/keyrings/docker.gpg

# Añadir el repositorio de Docker (necesario para Containerd)
echo \
  "deb [arch=$(dpkg --print-architecture) signed-by=/etc/apt/keyrings/docker.gpg] https://download.docker.com/linux/ubuntu \
  $(. /etc/os-release && echo "$VERSION_CODENAME") stable" | \
  sudo tee /etc/apt/sources.list.d/docker.list > /dev/null

# Actualizar la lista de paquetes con el nuevo repositorio
sudo apt update

# Instalar Containerd
sudo apt install -y containerd.io

# Configurar Containerd para usar cgroup driver de systemd (crucial para K8s)
sudo mkdir -p /etc/containerd
sudo containerd config default | sudo tee /etc/containerd/config.toml
sudo sed -i 's/SystemdCgroup = false/SystemdCgroup = true/g' /etc/containerd/config.toml

# Reiniciar y habilitar Containerd
sudo systemctl restart containerd
sudo systemctl enable containerd

# Añadir la clave GPG de Kubernetes
curl -fsSL https://pkgs.k8s.io/core:/stable:/v1.28/deb/Release.key | sudo gpg --dearmor -o /etc/apt/keyrings/kubernetes-apt-keyring.gpg
sudo chmod a+r /etc/apt/keyrings/kubernetes-apt-keyring.gpg

# Añadir el repositorio de Kubernetes
echo 'deb [signed-by=/etc/apt/keyrings/kubernetes-apt-keyring.gpg] https://pkgs.k8s.io/core:/stable:/v1.28/deb/ /' | sudo tee /etc/apt/sources.list.d/kubernetes.list

# Actualizar la lista de paquetes
sudo apt update

# Instalar kubeadm, kubelet, kubectl
sudo apt install -y kubeadm kubelet kubectl

# Bloquear las versiones para evitar actualizaciones automáticas y problemas
sudo apt-mark hold kubeadm kubelet kubectl
